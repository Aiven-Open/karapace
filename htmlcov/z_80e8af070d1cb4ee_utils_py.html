115def canonicalize_draft_keywords(schema: dict) -> dict: 

116 """Rewrite Draft 2019-09 / 2020-12 keywords into their Draft-7 equivalents. 

117 

118 The compatibility engine is expressed in terms of Draft-7 keyword shapes. Rewriting the 

119 renamed/relocated keywords here means every downstream check works unchanged, and comparisons 

120 between schemas written in different drafts become well-defined (a single canonical form). 

121 Keywords that cannot be canonicalized safely are handled by a fail-closed guard before 

122 compatibility checking, not here. 

123 """ 

124 # 2020-12 array model: `prefixItems` holds the positional (tuple) schemas that Draft-7 spells as 

125 # the list form of `items`; when `prefixItems` is present, `items` holds the "additional items" 

126 # schema that Draft-7 spells as `additionalItems`. 

127 if PREFIX_ITEMS in schema: 

128 prefix_items = schema.pop(PREFIX_ITEMS) 

129 additional_items = schema.pop(Keyword.ITEMS.value, None) 

130 schema[Keyword.ITEMS.value] = prefix_items 

131 if additional_items is not None and Keyword.ADDITIONAL_ITEMS.value not in schema: 

132 schema[Keyword.ADDITIONAL_ITEMS.value] = additional_items 

133 

134 # 2019-09 rename: `$defs` is the new name for `definitions`. References are already resolved by 

135 # normalize_schema_rec, so this only keeps the canonical key name for any surviving subschema. 

136 if DEFS in schema and DEFINITIONS not in schema: 

137 schema[DEFINITIONS] = schema.pop(DEFS) 

138 

139 # 2019-09 split: `dependentRequired` is the string-array form of Draft-7 `dependencies`. 

140 if DEPENDENT_REQUIRED in schema: 

141 dependent_required = schema.pop(DEPENDENT_REQUIRED) 

142 existing = schema.get(Keyword.DEPENDENCIES.value, {}) 

143 if isinstance(existing, dict) and isinstance(dependent_required, dict): 143 ↛ 147line 143 didn't jump to line 147 because the condition on line 143 was always true

144 # Existing `dependencies` entries win on key conflict. 

145 schema[Keyword.DEPENDENCIES.value] = {**dependent_required, **existing}

147 return schema

150def find_unsupported_compat_keywords(schema: Any) -> set[str]: 

151 """Recursively collect keywords (in keyword position) the compatibility engine cannot evaluate. 

152 

153 Returns the subset of ``UNSUPPORTED_COMPAT_KEYWORDS`` used as *schema keywords* anywhere in the 

154 schema. Used by the fail-closed guard before normalization, so it runs on the original schema. 

155 

156 The walk is position-aware to avoid false positives: a keyword-like string is only a match when 

157 it is a schema keyword, not when it is an author-chosen name (e.g. a property named 

158 ``unevaluatedItems``) or a value inside instance-data keywords (``enum``, ``const``, ``default``). 

159 """ 

160 found: set[str] = set() 

161 if isinstance(schema, dict): 

162 for key, value in schema.items(): 

163 if key in UNSUPPORTED_COMPAT_KEYWORDS: 

164 found.add(key) 

165 

166 if key in _DATA_VALUE_KEYWORDS: 

167 # Value is arbitrary instance data, never a subschema — do not descend. 

168 continue 

169 if key in _NAME_KEYED_MAP_KEYWORDS and isinstance(value, dict): 

170 # Keys here are author-chosen names; only the values are subschemas. 

171 for subschema in value.values(): 

172 found |= find_unsupported_compat_keywords(subschema) 

173 else: 

174 found |= find_unsupported_compat_keywords(value) 

175 elif isinstance(schema, list): 175 ↛ 176line 175 didn't jump to line 176 because the condition on line 175 was never true

176 for item in schema: 

177 found |= find_unsupported_compat_keywords(item)
